                    </div>

                grid.appendChild(card);
            });
        }

        function startOvenBaking() {
            goToStep('4-oven');
            let progress = 0;
            const progressBar = document.getElementById('timerProgress');
            const statusText = document.getElementById('ovenStatusText');
            const ovenCanvas = document.getElementById('canvasOven');
            const ovenCtx = ovenCanvas.getContext('2d');

            let bakeInterval = setInterval(() => {
                progress += 2;
                progressBar.style.width = progress + '%';

                let bfactor = progress / 100;
                let bakedColor = interpolateColor("#ffdf6d", mooncakeData.color, bfactor);
                drawMooncakeGraphic(ovenCtx, 130, 130, bakedColor, mooncakeData.fillingId, mooncakeData.patternId, mooncakeData.toppings);

                if (progress < 40) statusText.innerText = "Baking aroma filling the kitchen...";
                else if (progress < 80) statusText.innerText = "Crust turning golden brown & crisp...";
                else statusText.innerText = "Ding! Mooncake baked to perfection! ✨";

                if (progress >= 100) {
                    clearInterval(bakeInterval);
                    playMagicDing();
                    setTimeout(() => { goToStep(5); }, 900);
                }
            }, 60);
        }

        function updateAllCanvases() {
            ['canvas1', 'canvas2', 'canvas3', 'canvasToppings'].forEach(id => {
                const c = document.getElementById(id);
                if(c) {
                    drawMooncakeGraphic(c.getContext('2d'), c.width, c.height, mooncakeData.color, mooncakeData.fillingId, mooncakeData.patternId, mooncakeData.toppings);
                }
            });
        }

        // --- Photorealistic Canvas Rendering Engine ---
        function drawMooncakeGraphic(ctx, width, height, crustColor, fillingId, patternId, toppings, isThumb = false) {
            ctx.clearRect(0, 0, width, height);
            const cx = width / 2;
            const cy = height / 2;
            const radius = (width / 2) - (isThumb ? 2 : 10);

            ctx.save();
            ctx.translate(cx, cy);

            // 1. Realistic 3D Baked Scalloped Edge
            const petals = isThumb ? 10 : 18;
            ctx.fillStyle = shadeColor(crustColor, -30);
            ctx.beginPath();
            for (let i = 0; i < petals; i++) {
                let angle = (i * 2 * Math.PI) / petals;
                let x = Math.cos(angle) * radius;
                let y = Math.sin(angle) * radius;
                if (i === 0) ctx.moveTo(x, y);
                else ctx.arc(x, y, radius * 0.15, angle - Math.PI/2, angle + Math.PI/2);
            }
            ctx.fill();

            // 2. Main Crust with Photorealistic Radial Sheen
            let crustGrad = ctx.createRadialGradient(-radius*0.2, -radius*0.2, radius*0.1, 0, 0, radius);
            crustGrad.addColorStop(0, shadeColor(crustColor, 25));
            crustGrad.addColorStop(0.5, crustColor);
            crustGrad.addColorStop(1, shadeColor(crustColor, -35));

            ctx.fillStyle = crustGrad;
            ctx.beginPath();
            ctx.arc(0, 0, radius * 0.91, 0, 2 * Math.PI);
            ctx.fill();

            // 3. Inner Filling Ring (Realistic cut or center preview)
            let fillData = fillings[fillingId];
            let fillGrad = ctx.createRadialGradient(0, 0, 2, 0, 0, radius * 0.65);
            fillGrad.addColorStop(0, shadeColor(fillData.color, 20));
            fillGrad.addColorStop(1, fillData.color);

            ctx.fillStyle = fillGrad;
            ctx.beginPath();
            ctx.arc(0, 0, radius * 0.72, 0, 2 * Math.PI);
            ctx.fill();

            // If Salted Egg Yolk filling, draw realistic embedded central yellow egg yolk
            if (fillingId === 3) {
                ctx.fillStyle = '#ff9900';
                ctx.beginPath();
                ctx.arc(0, 0, radius * 0.28, 0, 2 * Math.PI);
                ctx.fill();
                ctx.fillStyle = '#ffcc00';
                ctx.beginPath();
                ctx.arc(-radius*0.06, -radius*0.06, radius * 0.1, 0, 2 * Math.PI);
                ctx.fill();
            }

            // Crust Surface Pressed Ring
            ctx.strokeStyle = shadeColor(crustColor, -45);
            ctx.lineWidth = isThumb ? 0.8 : 2;
            ctx.beginPath();
            ctx.arc(0, 0, radius * 0.82, 0, 2 * Math.PI);
            ctx.stroke();

            // 4. Photorealistic Embossed Pattern Mold
            ctx.strokeStyle = shadeColor(crustColor, -55);
            ctx.fillStyle = shadeColor(crustColor, -45);
            ctx.lineWidth = isThumb ? 0.8 : 1.8;
            drawPatternDetails(ctx, patternId, radius * 0.55);

            // 5. Realistic Toppings
            if (toppings && toppings.sesame) {
                ctx.fillStyle = '#ffffff';
                for(let i=0; i<6; i++) {
                    let angle = i * Math.PI / 3 + 0.2;
                    let tx = Math.cos(angle) * radius * 0.38;
                    let ty = Math.sin(angle) * radius * 0.38;
                    ctx.beginPath(); ctx.ellipse(tx, ty, 2.5, 1.8, angle, 0, 2*Math.PI); ctx.fill();
                }
            }
            if (toppings && toppings.peanuts) {
                ctx.fillStyle = '#7a4215';
                for(let i=0; i<5; i++) {
                    let angle = i * Math.PI / 2.5;
                    let tx = Math.cos(angle) * radius * 0.32;
                    let ty = Math.sin(angle) * radius * 0.32;
                    ctx.beginPath(); ctx.arc(tx, ty, 3, 0, 2*Math.PI); ctx.fill();
                }
            }
            if (toppings && toppings.gold) {
                ctx.fillStyle = '#ffe83b';
                for(let i=0; i<4; i++) {
                    let angle = i * Math.PI / 2 + 0.5;
                    let tx = Math.cos(angle) * radius * 0.22;
                    let ty = Math.sin(angle) * radius * 0.22;
                    ctx.fillRect(tx, ty, 3.5, 3.5);
                }
            }

            ctx.restore();
        }

        function drawPatternDetails(ctx, patternId, r) {
            switch(patternId) {
                case 0:
                    for (let i = 0; i < 6; i++) {
                        ctx.save(); ctx.rotate((i * Math.PI) / 3);
                        ctx.beginPath(); ctx.ellipse(0, -r * 0.4, r * 0.18, r * 0.35, 0, 0, 2 * Math.PI); ctx.stroke();
                        ctx.restore();
                    }
                    ctx.beginPath(); ctx.arc(0, 0, r * 0.15, 0, 2*Math.PI); ctx.stroke();
                    break;
                case 1:
                    ctx.beginPath();
                    for (let i = 0; i < 5; i++) {
                        let angle = (i * 4 * Math.PI) / 5 - Math.PI / 2;
                        let x = Math.cos(angle) * r * 0.68; let y = Math.sin(angle) * r * 0.68;
                        if (i === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y);
                    }
                    ctx.closePath(); ctx.stroke();
                    break;
                case 2:
                    for (let i = 0; i < 5; i++) {
                        ctx.save(); ctx.rotate((i * 2 * Math.PI) / 5);
                        ctx.beginPath(); ctx.arc(0, -r * 0.35, r * 0.25, 0, Math.PI); ctx.stroke();
                        ctx.restore();
                    }
                    ctx.beginPath(); ctx.arc(0, 0, r * 0.25, 0, 2*Math.PI); ctx.stroke();
                    break;
                case 3:
                    for (let i = 0; i < 5; i++) {
                        ctx.save(); ctx.rotate((i * 2 * Math.PI) / 5);
                        ctx.beginPath(); ctx.ellipse(0, -r * 0.38, r * 0.2, r * 0.32, 0, 0, 2*Math.PI); ctx.stroke();
                        ctx.restore();
                    }
                    ctx.beginPath(); ctx.arc(0, 0, r * 0.12, 0, 2*Math.PI); ctx.fill();
                    break;
                case 4:
                    ctx.beginPath(); ctx.arc(0, 0, r * 0.58, 0, 2*Math.PI); ctx.stroke();
                    ctx.strokeRect(-r * 0.28, -r * 0.28, r * 0.56, r * 0.56);
                    break;
                case 5:
                    ctx.beginPath(); ctx.arc(0, 0, r * 0.6, 0, 2*Math.PI); ctx.stroke();
                    ctx.beginPath(); ctx.moveTo(-r * 0.32, 0); ctx.lineTo(r * 0.32, 0);
                    ctx.moveTo(0, -r * 0.32); ctx.lineTo(0, r * 0.32); ctx.stroke();
                    break;
            }
        }

        // Sticker touch/drag support
        function addSticker(emoji) {
            playMagicDing();
            const letterBox = document.getElementById('paperLetterBox');
            const sticker = document.createElement('div');
            sticker.className = 'placed-sticker';
            sticker.innerText = emoji;
            sticker.style.left = (25 + Math.random() * 260) + 'px';
            sticker.style.top = (35 + Math.random() * 160) + 'px';
            
            let isDragging = false;
            sticker.onpointerdown = (e) => { isDragging = true; sticker.setPointerCapture(e.pointerId); };
            sticker.onpointermove = (e) => {
                if (!isDragging) return;
                const rect = letterBox.getBoundingClientRect();
                sticker.style.left = (e.clientX - rect.left - 20) + 'px';
                sticker.style.top = (e.clientY - rect.top - 20) + 'px';
            };
            sticker.onpointerup = () => { isDragging = false; };
            letterBox.appendChild(sticker);
        }

        // --- Web Audio API SFX & Music ---
        let audioCtx = null;
        function getAudioContext() {
            if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            if (audioCtx.state === 'suspended') audioCtx.resume();
            return audioCtx;
        }

        function playSquishSound() {
            const ctx = getAudioContext();
            let osc = ctx.createOscillator(), gain = ctx.createGain();
            osc.type = 'triangle';
            osc.frequency.setValueAtTime(170, ctx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(70, ctx.currentTime + 0.14);
            gain.gain.setValueAtTime(0.18, ctx.currentTime);
            gain.gain.linearRampToValueAtTime(0.01, ctx.currentTime + 0.14);
            osc.connect(gain); gain.connect(ctx.destination);
            osc.start(); osc.stop(ctx.currentTime + 0.14);
        }

        function playStampSound() {
            const ctx = getAudioContext();
            let osc = ctx.createOscillator(), gain = ctx.createGain();
            osc.type = 'square';
            osc.frequency.setValueAtTime(280, ctx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(90, ctx.currentTime + 0.09);
            gain.gain.setValueAtTime(0.12, ctx.currentTime);
            gain.gain.linearRampToValueAtTime(0.01, ctx.currentTime + 0.09);
            osc.connect(gain); gain.connect(ctx.destination);
            osc.start(); osc.stop(ctx.currentTime + 0.09);
        }

        function playMagicDing() {
            const ctx = getAudioContext();
            let now = ctx.currentTime;
            [523.25, 659.25, 783.99, 1046.50].forEach((freq, idx) => {
                let osc = ctx.createOscillator(), gain = ctx.createGain();
                osc.type = 'sine';
                osc.frequency.setValueAtTime(freq, now + idx * 0.07);
                gain.gain.setValueAtTime(0.1, now + idx * 0.07);
                gain.gain.exponentialRampToValueAtTime(0.0001, now + idx * 0.07 + 0.35);
                osc.connect(gain); gain.connect(ctx.destination);
                osc.start(now + idx * 0.07); osc.stop(now + idx * 0.07 + 0.35);
            });
        }

        let isPlaying = false, songInterval = null;
        const notes = [261.63, 293.66, 329.63, 392.00, 440.00, 523.25];
        const melody = [0, 2, 4, 3, 2, 0, 1, 3, 4, 5, 4, 2];
        let noteIdx = 0;

        function toggleAudio() {
            const btn = document.getElementById('audioToggleBtn');
            const ctx = getAudioContext();
            if (!isPlaying) {
                if (songInterval) clearInterval(songInterval);
                songInterval = setInterval(() => {
                    let osc = ctx.createOscillator(), gain = ctx.createGain();
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(notes[melody[noteIdx]], ctx.currentTime);
                    gain.gain.setValueAtTime(0.05, ctx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime + 0.45);
                    osc.connect(gain); gain.connect(ctx.destination);
                    osc.start(); osc.stop(ctx.currentTime + 0.45);
                    noteIdx = (noteIdx + 1) % melody.length;
                }, 580);
                isPlaying = true; btn.innerText = "Pause Music";
            } else {
                clearInterval(songInterval); songInterval = null;
                isPlaying = false; btn.innerText = "Play Music";
            }
        }

        function shadeColor(color, percent) {
            let num = parseInt(color.replace("#",""), 16),
                amt = Math.round(2.55 * percent),
                R = (num >> 16) + amt, G = (num >> 8 & 0x00FF) + amt, B = (num & 0x0000FF) + amt;
            return "#" + (0x1000000 + (R<255?R<0?0:R:255)*0x10000 + (G<255?G<0?0:G:255)*0x100 + (B<255?B<0?0:B:255)).toString(16).slice(1);
        }

        function interpolateColor(color1, color2, factor) {
            let c1 = parseInt(color1.replace('#',''), 16), c2 = parseInt(color2.replace('#',''), 16);
            let r = Math.round((c1 >> 16) + factor * ((c2 >> 16) - (c1 >> 16)));
            let g = Math.round(((c1 >> 8) & 0xff) + factor * (((c2 >> 8) & 0xff) - ((c1 >> 8) & 0xff)));
            let b = Math.round((c1 & 0xff) + factor * ((c2 & 0xff) - (c1 & 0xff)));
            return `rgb(${r},${g},${b})`;
        }
